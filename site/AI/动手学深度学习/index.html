      import torch
x = torch.arange(4.0, requires_grad=True) # tensor([0., 1., 2., 3.], requires_grad=True)
y = x * x                                 # tensor([0., 1., 4., 9.], grad_fn=<MulBackward0>)
y.sum()                                   # tensor(14., grad_fn=<SumBackward0>)
y.sum().backward()                        # y.backward() 报错：grad can be implicitly created only for scalar outputs
x.grad                                    # tensor([0., 2., 4., 6.])

      y 是一维张量（shape=[4]），不是标量。backward() 默认只能对标量张量直接反向传播；多维张量执行backward()必须传入梯度权重参数gradient，否则直接报错。

      数学：\(L=\sum y_i\)，\(\dfrac{dL}{dx_i}=2x_i\)，刚好拿到每个\(x_i\)的导数。深度学习绝大多数 loss 都是求和 / 平均，这是工业常规写法。

      import random
import torch
from d2l import torch as d2l

def synthetic_data(w, b, num_examples):
     """生成y=Xw+b+噪声"""
     X = torch.normal(0, 1, (num_examples, len(w)))
     y = torch.matmul(X, w) + b
     y += torch.normal(0, 0.01, y.shape)
     return X, y.reshape((-1, 1))
     # reshape 中 -1 = 自动推导该维度长度，所以这里是指列为1

true_w = torch.tensor([2, -3.4])
true_b = 4.2
features, labels = synthetic_data(true_w, true_b, 1000)

def data_iter(batch_size, features, labels):
     num_examples = len(features)
     indices = list(range(num_examples))
     random.shuffle(indices)
     for i in range(0, num_examples, batch_size):
         batch_indices = torch.tensor(
             indices[i: min(i + batch_size, num_examples)])
         yield features[batch_indices], labels[batch_indices]

batch_size = 10

for X, y in data_iter(batch_size, features, labels):
     print(X, '\n', y)
     break

w = torch.normal(0, 0.01, size=(2,1), requires_grad=True)
b = torch.zeros(1, requires_grad=True)

def linreg(X, w, b):
     """线性回归模型"""
     return torch.matmul(X, w) + b

def squared_loss(y_hat, y):
     """均方损失"""
     return (y_hat - y.reshape(y_hat.shape)) ** 2 / 2

def sgd(params, lr, batch_size):
     """小批量随机梯度下降"""
     with torch.no_grad():
         for param in params:
             param -= lr * param.grad / batch_size
             param.grad.zero_()

lr = 0.03
num_epochs = 3
net = linreg
loss = squared_loss

for epoch in range(num_epochs):
     for X, y in data_iter(batch_size, features, labels):
         l = loss(net(X, w, b), y)
         l.sum().backward()
         sgd([w, b], lr, batch_size)
     with torch.no_grad():
         train_l = loss(net(features, w, b), labels)
         print(f'epoch {epoch + 1}, loss {float(train_l.mean()):f}')

print(f'w的估计误差: {true_w - w.reshape(true_w.shape)}')
print(f'b的估计误差: {true_b - b}')

def load_array(data_arrays, batch_size, is_train=True):  
    """构造一个PyTorch数据迭代器"""
    dataset = data.TensorDataset(*data_arrays)
    return data.DataLoader(dataset, batch_size, shuffle=is_train)

batch_size = 10
data_iter = load_array((features, labels), batch_size)

next(iter(data_iter))

from torch import nn

net = nn.Sequential(nn.Linear(2, 1))

net[0].weight.data.normal_(0, 0.01)
net[0].bias.data.fill_(0)

loss = nn.MSELoss()

trainer = torch.optim.SGD(net.parameters(), lr=0.03)

num_epochs = 3
for epoch in range(num_epochs):
    for X, y in data_iter:
        l = loss(net(X) ,y)
        trainer.zero_grad()
        l.backward()
        trainer.step()

    Softmax回归¶

    1. Softmax回归

      Softmax 多用于互斥多分类输出层（一张图只能是猫 / 狗 / 鸟其中一类），也是 Transformer 注意力机制的核心归一化操作。专门把任意实数向量（神经网络原始打分，称为 Logits，即\(z_i\)），转换成合法概率分布：

      • 所有输出值落在 \((0,1)\)；
      • 全部元素相加总和 = 1；
      • 输入分数越高，对应概率越大，自带类别竞争关系。

      公式：设输入向量：\(\boldsymbol z = [z_1,z_2,...,z_K]\)，共 K 个类别。 对第 i 个分量： \(s_i=\text{softmax}(z_i)=\frac{e^{z_i}}{\sum_{j=1}^K e^{z_j}}\)， e 是自然指数；分母是所有指数求和，做全局归一化。

      1）信息论原始交叉熵公式

      对于单个样本： 模型预测概率分布 p，真实标签分布 y（one-hot） \(CE(y,p) = -\sum_{i=1}^K y_i \log(p_i)\)

      • K：总类别数
      • \(y_i\)：真实 one-hot，只有正确类别为 1，其余全 0
      • \(p_i\)：模型给出第 i 类概率，必须满足 \(0<p_i<1、\sum p_i=1\)

      因为 one-hot 只有正确类别位置是 1，求和只剩一项： 假设真实类别是第 t 类，则 \(y_t=1\)，其余\(y_i=0\) 👉 \(CE = -\log(p_t)\)

      含义：只惩罚「正确类别的预测概率」。正确类别概率越高，损失越小；概率越低，损失爆炸变大。举例子：3 分类，真实标签是类别 2

      • 模型算出正确类概率 \(p_2=0.9\)，损失 \(-\log(0.9)\approx0.105\)（损失很小）
      • 模型算出正确类概率 \(p_2=0.1\)，损失 \(-\log(0.1)\approx2.303\)（损失很大）

      2）Softmax 与交叉熵的绑定关系

      Softmax：接收 logits，把无界分数转为合法概率 \(p_i=\dfrac{e^{z_i}}{\sum e^{z_j}}\)，保证总和 = 1、值域 (0,1)，满足交叉熵对输入必须是概率的硬性要求。 没有 softmax，logits 可正可负，log 有可能取负数、无意义。

      交叉熵：拿到 softmax 输出的概率，计算预测和真实标签的差距，作为 loss 用来反向传播更新权重。

      串联流程： \(\text{网络输出logits } z \xrightarrow{\text{Softmax}} 概率p \xrightarrow{\text{交叉熵}} loss\)

      3）PyTorch 设计：合并为 CrossEntropyLoss

      PyTorch 不会让你手动先 softmax 再算 loss，而是公式合并： 把 \(p_i=\dfrac{e^{z_i}}{\sum e^{z_j}}\) 代入交叉熵公式： \(CE = -\log\left( \frac{e^{z_t}}{\sum_{j}e^{z_j}} \right) = -z_t + \log\left(\sum_j e^{z_j}\right)\)

      4）梯度推导

      先把所有符号先约定清楚：

      • \(z_i\)：神经网络输出的 logits（未经过 softmax 的原始分数）

      • \(p_i=\mathrm{softmax}(z_i)=\dfrac{e^{z_i}}{\sum\limits_{j}e^{z_j}}\)，softmax‑输出，代表预测概率

      • 标签 \(y_i\)：独热编码，只有正确类别 t 处 \(y_t=1\)，其余 \(y_{i\neq t}=0\)

      • 单个样本交叉熵损失：

        \(CE = -\log(p_t)=-\log\left(\frac{e^{z_t}}{\sum_j e^{z_j}}\right) =-z_t+\log\Big(\sum_j e^{z_j}\Big)\)

      梯度推导分两种情况，\(i=t\)（正确类别）和 \(i\neq t\)（其他类别）

      损失函数：

      \(L=-z_t+\log\sum_{j}e^{z_j}\)

      • 情况 1：求偏导 \(\boldsymbol{i = t}\)

        \(\frac{\partial L}{\partial z_t} =\frac{\partial(-z_t)}{\partial z_t} +\frac{\partial}{\partial z_t}\log\Big(\sum_j e^{z_j}\Big)\)

        第一项：\(\dfrac{\partial(-z_t)}{\partial z_t}=-1\)

        第二项链式法则： \(\frac{\partial}{\partial z_t}\log\Big(\sum_j e^{z_j}\Big) =\dfrac{\dfrac{\partial}{\partial z_t}\big(\sum e^{z_j}\big)}{\sum e^{z_j}} =\frac{e^{z_t}}{\sum e^{z_j}}=p_t\)

        合并： \(\frac{\partial L}{\partial z_t}= -1+p_t\)

        因为独热标签 \(y_t=1\)： \(\frac{\partial L}{\partial z_t}=p_t-1=\boldsymbol{p_t-y_t}\)

      • 情况 2：\(\boldsymbol{i\neq t}\)（不是正确标签）

        此时损失式子里 \(-z_t\) 和 \(z_i\) 无关，导数等于 0

        \(\frac{\partial L}{\partial z_i} =0+\frac{\partial}{\partial z_i}\log\Big(\sum_j e^{z_j}\Big) =\frac{e^{z_i}}{\sum e^{z_j}}=p_i\)

        又因为 \(i\neq t\)，独热编码 \(y_i=0\)

        \(\frac{\partial L}{\partial z_i}=p_i-0=\boldsymbol{p_i-y_i}\)

      合并两种结果不管 i 是不是目标类别： \(\boldsymbol{\frac{\partial loss}{\partial z_i}=p_i-y_i}\)

      总结

      • softmax 把 logits‑z 转为概率 p
      • 交叉熵对原始输出 z 的梯度非常简洁：预测概率 − one‑hot 标签
      • 也是 PyTorch CrossEntropyLoss 底层原理：内部先做 softmax，再算交叉熵，梯度直接是 \(p-y\)，不用手动搭建 softmax 层。

      直观解释

      • 当预测越不准：\(p_t\) 越小，梯度 \(p_t-1\) 负数值越大，参数更新幅度越强；
      • 预测完全正确 \(p_t=1\)，梯度等于 0，权重不再更新。
