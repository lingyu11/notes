    第6章 完善内核¶

    在上一章中，我们成功构建了一个简易内核，功能较为基础。

    本章将深入完善内核功能，重点介绍函数调用约定、汇编与C语言混合编程、实现关键打印函数,为内核开发奠定坚实基础。

    6.1 函数调用约定简介¶

    在混合编程中，汇编与C代码相互调用需遵循统一的接口规范，即函数调用约定（Calling Conventions）。

    调用约定的核心要明确了以下细节：

    • 参数传递方式：通过寄存器或栈传递。

    • 参数传递顺序：从左到右或从右到左。

    • 寄存器环境恢复：由调用者或被调用者负责。

    例如，C函数调用subtract(3, 2)，若在汇编层需解决参数存储顺序和栈空间清理问题。

    本节将解析参数传递、栈清理等关键机制，并通过实例对比cdecl与stdcall约定。

    C语言遵守的是cdecl，以下通过对比cdecl与stdcall约定说明其差异。

    cdecl约定示例

    • 参数从右向左入栈，调用者清理栈空间。
    • 支持变长参数，如printf函数。

    stdcall 从右到左 被调用者 Windows API常用

    6.2 汇编语言和C语言混合编程¶

    混合编程是内核开发的核心技术，本节将讲述库函数与系统调用的关系，并用实例展示汇编与C代码相互调用的实现。

    6.2.1 浅析C库函数与系统调用¶

    系统调用是内核提供的服务接口。应用程序可通过C库函数间接使用系统调用：

    1
2
3
4

    #include <unistd.h>
int main() {
    write(1, "hello\n", 6);  // 依赖C库
    return 0;
}

    间接使用库函数进行系统调用最终还是需要软中断int 0x80请求内核功能，,write系统调用功能号4，只是库函数封装了将软中断int 0x80。用户程序可以跨过库函数，直接与系统内核通信。

int 0x80               ;发起中断，通知 Linux 完成请求的功能

global asm_print

asm_print:
    push ebp
    mov ebp, esp
    mov ebx, 1
    mov ecx, [ebp+8] ; 第一个参数
    mov edx, [ebp+12] ; 第二个参数
    mov eax, 4       ; write系统调用

void c_print(char* str) {

    6.3.1 显卡的端口控制¶

    寄存器组 地址寄存器端口 数据寄存器端口 说明
    CRT控制器 0x3D4 0x3D5 控制光标位置、滚屏等

    04h Start Horizontal Retrace Register 11h Vertical Retrace End Register

section .text
global put_char

put_char:
    pushad                     ; 备份寄存器
    mov ax, SELECTOR_VIDEO     ; 设置视频段选择子
    mov gs, ax

    ; 获取光标位置（高8位）
    mov dx, 0x3D4
    mov al, 0x0E
    out dx, al
    mov dx, 0x3D5
    in al, dx
    mov ah, al

    ; 获取光标位置（低8位）
    mov dx, 0x3D4
    mov al, 0x0F
    out dx, al
    mov dx, 0x3D5
    in al, dx

    ; 光标值存入bx
    mov bx, ax
    mov ecx, [esp+36]          ; 获取传入的字符

    ; 判断字符类型
    cmp cl, 0x0D               ; 回车符
    jz .carriage_return
    cmp cl, 0x0A               ; 换行符
    jz .line_feed
    cmp cl, 0x08               ; 退格符
    jz .backspace

    jmp .printable

;退格
.backspace:
    dec bx                     ; 光标前移
    shl bx, 1                  ; 转换为显存偏移
    mov byte [gs:bx], 0x20     ; 写入空格
    inc bx
    mov byte [gs:bx], 0x07     ; 属性字节
    shr bx, 1
    jmp .set_cursor

;打印可见字符
.printable:
    shl bx, 1
    mov [gs:bx], cl            ; 写入字符
    inc bx
    mov byte [gs:bx], 0x07     ; 属性字节
    shr bx, 1
    inc bx                     ; 光标后移
    cmp bx, 2000               ; 是否需滚屏
    jl .set_cursor

;回车换行处理
.line_feed:
.carriage_return:
    ; 处理换行/回车：光标移到下一行首
    xor dx, dx
    mov ax, bx
    mov si, 80
    div si                     ; bx/80，商在ax，余数在dx
    sub bx, dx                 ; 回车：回到行首
    add bx, 80                 ; 换行：移到下一行
    cmp bx, 2000
    jl .set_cursor

;滚屏处理
.roll_screen:
    ; 滚屏：第1-24行内容上移，清空最后一行
    cld
    mov ecx, 960               ; 3840字节/4字节
    mov edi, 0xC00B8000        ; 第1行起始
    mov esi, 0xC00B80A0        ; 第2行起始
    rep movsd                  ; 复制内存
;滚屏时最后一行空白处理
    mov ebx, 3840              ; 最后一行起始偏移
    mov ecx, 80
.clear_line:
    mov word [gs:ebx], 0x0720  ; 黑底白字空格
    add ebx, 2
    loop .clear_line
    mov bx, 1920               ; 光标置最后一行首

;更新光标位置
.set_cursor:
    ; 更新光标位置寄存器
    mov dx, 0x3D4
    mov al, 0x0E
    out dx, al

    mov dx, 0x3D5
    mov al, bh
    out dx, al                 ; 写入高8位

    mov dx, 0x3D4
    mov al, 0x0F
    out dx, al

    mov dx, 0x3D5
    mov al, bl
    out dx, al                 ; 写入低8位

    popad                      ; 恢复寄存器
    ret

    6.4 内联汇编¶

    内联汇编允许在C代码中嵌入汇编指令，兼具高效性与灵活性。本节简单介绍AT&T语法与GCC内联汇编规范。

    6.4.1 什么是内联汇编¶

    内联汇编分为两类：

    • 基本内联汇编：直接嵌入汇编代码，简单但功能有限。

    • 扩展内联汇编：通过约束条件与C变量交互，功能强大。

    6.4.2 汇编语言AT&T语法简介¶

    AT&T语法与Intel语法主要区别：

    特性 Intel语法 AT&T语法
    操作数顺序 目标在左 源在左
    寄存器前缀 无 %前缀
    立即数前缀 无 $前缀
    内存引用 [eax] (%eax)

    示例：

mov eax, 32

; AT&T语法
movl 4(%ebx), %eax
mov $32, %eax

    6.4.3 基本内联汇编¶

    基本内联汇编格式简单，使用 asm [volatile] ("asm code")的形式，直接嵌入汇编代码。

    主要用于执行简单的汇编指令，功能有限，只能通过全局变量与C代码交互，不支持局部变量或精细的参数传递。

    6.4.4 扩展内联汇编¶

    扩展内联汇编采用 asm [volatile] ("asm code" : output : input : clobber/modify)的复杂格式，引入了输出、输入和clobber/modify部分。

    它支持约束（如寄存器约束 "a" 表示eax），以及占位符（如 %0）来精确控制操作数，允许与C变量交互，包括局部变量。
