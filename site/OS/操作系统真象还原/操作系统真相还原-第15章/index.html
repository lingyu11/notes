    第15章 系统交互¶

    在掌握了进程管理和内存管理的核心机制后，操作系统需要与用户建立交互通道。本章将深入探讨如何构建系统交互的核心组件，从进程创建的fork机制，到键盘输入处理，再到shell命令解析，逐步构建一个完整的用户交互环境。

    15.1 fork的原理与实现¶

    进程创建是操作系统的基础功能，fork作为Unix/Linux系统的核心机制，实现了进程的克隆与分叉。理解fork的原理是构建完整系统交互环境的第一步。

    15.1.1 什么是fork¶

    fork函数原型为 pid_t fork(void)，其独特之处在于一个函数调用可能返回三种不同值： - 父进程获得子进程的pid - 子进程获得0 - 失败时返回-1

    这种设计源于Linux中没有直接获取子进程pid的方法，因此fork通过返回值让父进程识别子进程身份，同时子进程可通过getppid()获取父进程pid。由于系统中不存在pid为0的进程，子进程返回0可有效区分父子进程。

    代码解析： - 第1-4行：程序先输出提示信息，休眠5秒为进程监控留出时间 - 第6行：fork调用是关键分叉点，执行后产生两个独立进程 - 第8-16行：根据pid值区分父子进程执行路径 - 父进程（pid>0）：输出自身pid - 子进程（pid=0）：输出自身pid - fork失败（pid=-1）：直接退出

    运行结果显示fork后系统中存在两个同名进程，父进程pid为2810，子进程pid为2815。这证明fork本质上是进程克隆机制，将原进程的完整程序体（代码段、数据段等）复制到新进程中，形成两个独立执行流。

void* get_a_page_without_opvaddrbitmap(enum pool_flags pf, uint32_t vaddr) {
    struct pool* mem_pool = pf & PF_KERNEL ? &kernel_pool : &user_pool;
    lock_acquire(&mem_pool->lock);
    void* page_phyaddr = palloc(mem_pool);
    if (page_phyaddr == NULL) {
        lock_release(&mem_pool->lock);
        return NULL;
    }
    page_table_add((void*)vaddr, page_phyaddr);
    lock_release(&mem_pool->lock);
    return (void*)vaddr;
}

static int32_t copy_pcb_vaddrbitmap_stack0(struct task_struct* child_thread, struct task_struct* parent_thread) {
    /* a. 复制pcb所在的整个页，里面包含进程pcb信息及特权级0的栈 */

    /* b. 单独修改关键属性 */

    /* 复制子进程的进程体(代码和数据)及用户栈 */
static void copy_body_stack3(struct task_struct* child_thread, struct task_struct* parent_thread, void* buf_page) {
    uint8_t* vaddr_btmp = parent_thread->userprog_vaddr.vaddr_bitmap.bits;
    uint32_t btmp_bytes_len = parent_thread->userprog_vaddr.vaddr_bitmap.btmp_bytes_len;
    uint32_t vaddr_start = parent_thread->userprog_vaddr.vaddr_start;
    uint32_t idx_byte = 0;
    uint32_t idx_bit = 0;
    uint32_t prog_vaddr = 0;

    /* 遍历父进程虚拟地址位图，找出已使用内存页 */
    while (idx_byte < btmp_bytes_len) {
        if (vaddr_btmp[idx_byte]) {
            idx_bit = 0;
            while (idx_bit < 8) {
                if ((BITMAP_MASK << idx_bit) & vaddr_btmp[idx_byte]) {
                    prog_vaddr = (idx_byte * 8 + idx_bit) * PG_SIZE + vaddr_start;

                    /* a. 将父进程用户空间数据复制到内核缓冲区 */
                    memcpy(buf_page, (void*)prog_vaddr, PG_SIZE);

                    /* b. 切换到子进程页表 */
                    page_dir_activate(child_thread);

                    /* c. 为子进程分配相同虚拟地址 */
                    get_a_page_without_opvaddrbitmap(PF_USER, prog_vaddr);

                    /* d. 从内核缓冲区复制到子进程用户空间 */
                    memcpy((void*)prog_vaddr, buf_page, PG_SIZE);

                    /* e. 恢复父进程页表 */
                    page_dir_activate(parent_thread);
                }
                idx_bit++;
            }
        }
        idx_byte++;
    }
}

    内存复制策略： - 按需复制：仅复制实际使用的内存页，而非整个3GB地址空间 - 内核中转：利用内核空间作为数据交换缓冲区 - 页表切换：确保内存分配操作在子进程地址空间进行 - 逐页处理：避免大块内存复制带来的性能问题

static int32_t build_child_stack(struct task_struct* child_thread) {

int32_t sys_read(int32_t fd, void* buf, uint32_t count) {
    ASSERT(buf != NULL);
    int32_t ret = -1;

    if (fd < 0 || fd == stdout_no || fd == stderr_no) {
        printk("sys_read: fd error\n");
    } else if (fd == stdin_no) {
        /* 处理标准输入（键盘） */
        char* buffer = buf;
        uint32_t bytes_read = 0;
        while (bytes_read < count) {
            *buffer = ioq_getchar(&kbd_buf);  // 从键盘缓冲区获取字符
            bytes_read++;
            buffer++;
        }
        ret = (bytes_read == 0 ? -1 : (int32_t)bytes_read);
    } else {
        /* 处理文件读取 */
        uint32_t _fd = fd_local2global(fd);
        ret = file_read(&file_table[_fd], buf, count);
    }
    return ret;
}

static int32_t cmd_parse(char* cmd_str, char** argv, char token) {
    ASSERT(cmd_str != NULL);
    int32_t arg_idx = 0;

    /* 初始化参数数组 */
    while (arg_idx < MAX_ARG_NR) {
        argv[arg_idx] = NULL;
        arg_idx++;
    }

    char* next = cmd_str;
    int32_t argc = 0;

    /* 外层循环：处理整个命令行 */
    while (*next) {
        /* 跳过前导空格 */
        while (*next == token) {
            next++;
        }
        if (*next == 0) break;  // 结束处理

        argv[argc] = next;  // 记录参数起始位置

        /* 内层循环：找到参数结束位置 */
        while (*next && *next != token) {
            next++;
        }

        /* 将分隔符替换为字符串结束符 */
        if (*next) {
            *next++ = 0;
        }

        argc++;
        if (argc >= MAX_ARG_NR) return -1;  // 参数过多
    }
    return argc;
}

    /* 路径清洗：将..和.转换为实际路径 */
static void wash_path(char* old_abs_path, char* new_abs_path) {
    ASSERT(old_abs_path[0] == '/');
    char name[MAX_FILE_NAME_LEN] = {0};
    char* sub_path = old_abs_path;

    sub_path = path_parse(sub_path, name);
    if (name[0] == 0) {  // 仅根目录
        new_abs_path[0] = '/';
        new_abs_path[1] = 0;
        return;
    }

    new_abs_path[0] = 0;
    strcat(new_abs_path, "/");

    while (name[0]) {
        if (!strcmp("..", name)) {
            /* 处理上级目录 */
            char* slash_ptr = strrchr(new_abs_path, '/');
            if (slash_ptr != new_abs_path) {
                *slash_ptr = 0;  // 去掉最后一级
            } else {
                *(slash_ptr + 1) = 0;  // 保持根目录
            }
        } else if (strcmp(".", name)) {
            /* 非当前目录，添加到路径 */
            if (strcmp(new_abs_path, "/")) {
                strcat(new_abs_path, "/");
            }
            strcat(new_abs_path, name);
        }
        memset(name, 0, MAX_FILE_NAME_LEN);
        if (sub_path) {
            sub_path = path_parse(sub_path, name);
        }
    }
}

/* 路径标准化：相对路径转绝对路径 */
void make_clear_abs_path(char* path, char* final_path) {
    char abs_path[MAX_PATH_LEN] = {0};

    if (path[0] != '/') {  // 相对路径
        if (getcwd(abs_path, MAX_PATH_LEN) != NULL) {
            if (!(abs_path[0] == '/' && abs_path[1] == 0)) {
                strcat(abs_path, "/");
            }
        }
    }
    strcat(abs_path, path);
    wash_path(abs_path, final_path);
}

    路径处理规则： - 相对路径：基于当前工作目录解析 - 绝对路径：直接处理，忽略当前目录 - 特殊符号：.表示当前目录，..表示上级目录 - 根目录保护：避免路径退到根目录以上

/* ls命令：列出目录内容 */
void buildin_ls(uint32_t argc, char** argv) {
    char* pathname = NULL;
    bool long_info = false;
    uint32_t arg_path_nr = 0;
    uint32_t arg_idx = 1;

    /* 解析参数 */
    while (arg_idx < argc) {
        if (argv[arg_idx][0] == '-') {
            if (!strcmp("-l", argv[arg_idx])) {
                long_info = true;
            } else if (!strcmp("-h", argv[arg_idx])) {
                printf("usage: -l list all information about the file.\n");
                return;
            } else {
                printf("ls: invalid option %s\n", argv[arg_idx]);
                return;
            }
        } else {
            if (arg_path_nr == 0) {
                pathname = argv[arg_idx];
                arg_path_nr = 1;
            } else {
                printf("ls: only support one path\n");
                return;
            }
        }
        arg_idx++;
    }

    /* 处理默认路径 */
    if (pathname == NULL) {
        if (NULL != getcwd(final_path, MAX_PATH_LEN)) {
            pathname = final_path;
        } else {
            printf("ls: getcwd for default path failed\n");
            return;
        }
    } else {
        make_clear_abs_path(pathname, final_path);
        pathname = final_path;
    }

    /* 获取文件状态 */
    struct stat file_stat;
    if (stat(pathname, &file_stat) == -1) {
        printf("ls: cannot access %s: No such file or directory\n", pathname);
        return;
    }

    /* 目录处理 */
    if (file_stat.st_filetype == FT_DIRECTORY) {
        struct dir* dir = opendir(pathname);
        struct dir_entry* dir_e = NULL;
        char sub_pathname[MAX_PATH_LEN] = {0};

        if (long_info) {
            printf("total: %d\n", file_stat.st_size);
            while ((dir_e = readdir(dir))) {
                char ftype = 'd';
                if (dir_e->f_type == FT_REGULAR) ftype = '-';

                memset(sub_pathname, 0, MAX_PATH_LEN);
                strcpy(sub_pathname, pathname);
                if (sub_pathname[strlen(sub_pathname)-1] != '/') {
                    strcat(sub_pathname, "/");
                }
                strcat(sub_pathname, dir_e->filename);

                memset(&file_stat, 0, sizeof(struct stat));
                if (stat(sub_pathname, &file_stat) == -1) continue;

                printf("%c %d %d %s\n", ftype, dir_e->i_no, file_stat.st_size, dir_e->filename);
            }
        } else {
            while ((dir_e = readdir(dir))) {
                printf("%s ", dir_e->filename);
            }
            printf("\n");
        }
        closedir(dir);
    } else {
        /* 文件处理 */
        if (long_info) {
            printf("- %d %d %s\n", file_stat.st_ino, file_stat.st_size, pathname);
        } else {
            printf("%s\n", pathname);
        }
    }
}

                strcpy(cwd_cache, new_path);
            }
        } else if (!strcmp("pwd", argv[0])) {
            buildin_pwd(argc, argv);
        } else if (!strcmp("ps", argv[0])) {
            buildin_ps(argc, argv);
        } else if (!strcmp("clear", argv[0])) {
            buildin_clear(argc, argv);
        } else if (!strcmp("mkdir", argv[0])) {
            buildin_mkdir(argc, argv);
        } else {
            printf("external command: %s\n", argv[0]);
        }
    }
}

   大小为filesz的段加载到虚拟地址为vaddr的内存 */
static bool segment_load(int32_t fd, uint32_t offset, 
                        uint32_t filesz, uint32_t vaddr) {
    uint32_t vaddr_first_page = vaddr & 0xfffff000;  // 获取页框起始地址
    uint32_t size_in_first_page = PG_SIZE - (vaddr & 0x00000fff);  // 第一页占用字节数
    uint32_t occupy_pages = 0;

    /* 计算需要的页框数 */
    if (filesz > size_in_first_page) {
        uint32_t left_size = filesz - size_in_first_page;
        occupy_pages = DIV_ROUND_UP(left_size, PG_SIZE) + 1;  // +1表示第一页
    } else {
        occupy_pages = 1;
    }

    /* 为进程分配内存 */
    uint32_t page_idx = 0;
    uint32_t vaddr_page = vaddr_first_page;
    while (page_idx < occupy_pages) {
        uint32_t* pde = pde_ptr(vaddr_page);
        uint32_t* pte = pte_ptr(vaddr_page);

        /* 如果页目录项或页表项不存在，分配内存 */
        if (!(( *pde & 0x00000001) && (*pte & 0x00000001))) {
            if (get_a_page(PF_USER, vaddr_page) == NULL) {
                return false;
            }
        }
        vaddr_page += PG_SIZE;
        page_idx++;
    }

    /* 从文件系统加载段到内存 */
    sys_lseek(fd, offset, SEEK_SET);
    sys_read(fd, (void*)vaddr, filesz);
    return true;
}

    /* 从文件系统上加载用户程序pathname，
   成功则返回程序的起始地址，否则返回-1 */
static int32_t load(const char* pathname) {
    int32_t ret = -1;
    struct Elf32_Ehdr elf_header;
    struct Elf32_Phdr prog_header;
    memset(&elf_header, 0, sizeof(struct Elf32_Ehdr));

    int32_t fd = sys_open(pathname, O_RDONLY);
    if (fd == -1) return -1;

    /* 读取ELF头 */
    if (sys_read(fd, &elf_header, sizeof(struct Elf32_Ehdr)) != 
        sizeof(struct Elf32_Ehdr)) {
        ret = -1;
        goto done;
    }

    /* 校验ELF头 */
    if (memcmp(elf_header.e_ident, "\177ELF\1\1\1", 7) ||  // 魔数校验
        elf_header.e_type != 2 ||        // 可执行文件
        elf_header.e_machine != 3 ||     // Intel 80386
        elf_header.e_version != 1 ||     // 版本1
        elf_header.e_phnum > 1024 ||     // 段数限制
        elf_header.e_phentsize != sizeof(struct Elf32_Phdr)) {  // 头大小
        ret = -1;
        goto done;
    }

    /* 遍历所有程序头 */
    Elf32_Off prog_header_offset = elf_header.e_phoff;
    Elf32_Half prog_header_size = elf_header.e_phentsize;
    uint32_t prog_idx = 0;

    while (prog_idx < elf_header.e_phnum) {
        memset(&prog_header, 0, prog_header_size);
        sys_lseek(fd, prog_header_offset, SEEK_SET);

        /* 读取程序头 */
        if (sys_read(fd, &prog_header, prog_header_size) != prog_header_size) {
            ret = -1;
            goto done;
        }

        /* 加载可加载段 */
        if (PT_LOAD == prog_header.p_type) {
            if (!segment_load(fd, prog_header.p_offset, 
                              prog_header.p_filesz, prog_header.p_vaddr)) {
                ret = -1;
                goto done;
            }
        }

        prog_header_offset += elf_header.e_phentsize;
        prog_idx++;
    }

    ret = elf_header.e_entry;  // 返回入口地址

done:
    sys_close(fd);
    return ret;
}

/* 用path指向的程序替换当前进程 */
int32_t sys_execv(const char* path, const char* argv[]) {
    uint32_t argc = 0;
    while (argv[argc]) argc++;  // 统计参数个数

    int32_t entry_point = load(path);
    if (entry_point == -1) return -1;  // 加载失败

    struct task_struct* cur = running_thread();
    memcpy(cur->name, path, TASK_NAME_LEN);  // 更新进程名
    cur->name[TASK_NAME_LEN-1] = 0;

    /* 准备新进程的内核栈 */
    struct intr_stack* intr_0_stack = (struct intr_stack*)
        ((uint32_t)cur + PG_SIZE - sizeof(struct intr_stack));

    intr_0_stack->ebx = (int32_t)argv;   // 参数数组地址
    intr_0_stack->ecx = argc;            // 参数个数
    intr_0_stack->eip = (void*)entry_point;  // 入口地址
    intr_0_stack->esp = (void*)0xc0000000;   // 用户栈顶

    /* 直接从中断返回，启动新进程 */
    asm volatile("movl %0, %%esp; jmp intr_exit" : : "g"(intr_0_stack) : "memory");
    return 0;  // 此行永远不会执行
}

    15.6 实现系统调用wait和exit¶

    进程生命周期管理需要完善的退出机制。本节将实现wait和exit系统调用，解决父子进程同步问题，回收僵尸进程资源，为用户程序提供标准退出接口。

    15.6.1 wait和exit的作用¶

    wait和exit是进程管理的核心系统调用： - exit：进程主动终止执行，向父进程返回状态值 - wait：父进程阻塞等待子进程结束，获取子进程返回状态

    两者配合解决了三个关键问题： 1. 进程同步：父进程等待子进程完成特定任务 2. 资源回收：内核回收子进程占用的内存、文件描述符等资源 3. 状态传递：子进程向父进程传递执行结果

    15.6.2 孤儿进程和僵尸进程¶

    孤儿进程：父进程先于子进程退出，子进程被init进程收养 僵尸进程：子进程退出后，父进程未调用wait获取其状态，导致PCB无法释放

static void release_prog_resource(struct task_struct* release_thread) {
    uint32_t* pgdir_vaddr = release_thread->pgdir;
    uint16_t user_pde_nr = 768, pde_idx = 0;

    /* 回收页表中的物理页 */
    while (pde_idx < user_pde_nr) {
        uint32_t* v_pde_ptr = pgdir_vaddr + pde_idx;
        uint32_t pde = *v_pde_ptr;

        if (pde & 0x00000001) {  // 页目录项存在
            uint32_t* first_pte_vaddr_in_pde = pte_ptr(pde_idx * 0x400000);
            uint16_t user_pte_nr = 1024, pte_idx = 0;

            /* 遍历页表项 */
            while (pte_idx < user_pte_nr) {
                uint32_t* v_pte_ptr = first_pte_vaddr_in_pde + pte_idx;
                uint32_t pte = *v_pte_ptr;

                if (pte & 0x00000001) {  // 页表项存在
                    uint32_t pg_phy_addr = pte & 0xfffff000;
                    free_a_phy_page(pg_phy_addr);
                }
                pte_idx++;
            }

            /* 回收页目录项对应的物理页 */
            uint32_t pg_phy_addr = pde & 0xfffff000;
            free_a_phy_page(pg_phy_addr);
        }
        pde_idx++;
    }

    /* 回收虚拟地址池位图 */
    uint32_t bitmap_pg_cnt = 
        release_thread->userprog_vaddr.vaddr_bitmap.btmp_bytes_len / PG_SIZE;
    uint8_t* user_vaddr_pool_bitmap = 
        release_thread->userprog_vaddr.vaddr_bitmap.bits;
    mfree_page(PF_KERNEL, user_vaddr_pool_bitmap, bitmap_pg_cnt);

    /* 关闭打开的文件 */
    uint8_t fd_idx = 3;
    while (fd_idx < MAX_FILES_OPEN_PER_PROC) {
        if (release_thread->fd_table[fd_idx] != -1) {
            sys_close(fd_idx);

        fd_idx++;

/* wait系统调用 */
pid_t sys_wait(int32_t* status) {
    struct task_struct* parent_thread = running_thread();

    while (1) {
        /* 优先处理已挂起的子进程 */
        struct list_elem* child_elem = list_traversal(&thread_all_list, 
                                                    find_hanging_child, 
                                                    parent_thread->pid);
        if (child_elem != NULL) {
            struct task_struct* child_thread = 
                elem2entry(struct task_struct, all_list_tag, child_elem);

            *status = child_thread->exit_status;  // 获取退出状态
            uint16_t child_pid = child_thread->pid;

            /* 回收子进程资源 */
            thread_exit(child_thread, false);
            return child_pid;
        }

        /* 检查是否有子进程 */
        child_elem = list_traversal(&thread_all_list, find_child, parent_thread->pid);
        if (child_elem == NULL) {  // 无子进程
            return -1;
        } else {  // 有子进程但未退出
            thread_block(TASK_WAITING);  // 阻塞父进程
        }
    }
}

/* exit系统调用 */
void sys_exit(int32_t status) {
    struct task_struct* child_thread = running_thread();
    child_thread->exit_status = status;  // 保存退出状态

    /* 将子进程过继给init */
    list_traversal(&thread_all_list, init_adopt_a_child, child_thread->pid);

    /* 回收资源（保留PCB） */
    release_prog_resource(child_thread);

    /* 唤醒等待的父进程 */
    struct task_struct* parent_thread = 
        pid2thread(child_thread->parent_pid);
    if (parent_thread->status == TASK_WAITING) {
        thread_unblock(parent_thread);
    }

    /* 挂起自己，等待父进程回收PCB */
    thread_block(TASK_HANGING);
}

    15.7 管道¶

    进程间通信是操作系统的重要功能。本节将实现管道机制，支持父子进程数据传递，并在shell中集成管道操作符，展示进程协作的完整解决方案。

    15.7.1 管道的原理¶

    管道是进程间通信的基础机制，其核心特点： - 文件抽象：管道在Linux中被视为特殊文件 - 环形缓冲区：内核中的内存缓冲区，大小固定 - 单向通信：数据只能单向流动（从写端到读端） - 父子共享：通过fork继承文件描述符实现共享

    管道工作流程： 1. 进程调用pipe()创建管道，获得两个文件描述符 2. 调用fork()创建子进程，继承管道描述符 3. 父进程关闭读端，子进程关闭写端（或反之） 4. 一方写入数据，另一方读取数据 5. 通信完成后关闭管道

    15.7.2 管道的设计¶

    设计要点： - 复用文件结构：利用现有文件系统框架 - 标志位识别：通过fd_flags特殊值(0xFFFF)标识管道 - 内存共享：fd_inode指向环形缓冲区内存 - 引用计数：fd_pos记录打开的描述符数量

uint32_t ioq_length(struct ioqueue* ioq) {
    uint32_t len = 0;
    if (ioq->head >= ioq->tail) {
        len = ioq->head - ioq->tail;
    } else {
        len = bufsize - (ioq->tail - ioq->head);
    }
    return len;
}

/* 创建管道 */
int32_t sys_pipe(int32_t pipefd[2]) {
    int32_t global_fd = get_free_slot_in_global();

    /* 分配环形缓冲区内存 */
    file_table[global_fd].fd_inode = get_kernel_pages(1);
    ioqueue_init((struct ioqueue*)file_table[global_fd].fd_inode);

    if (file_table[global_fd].fd_inode == NULL) {
        return -1;
    }

    /* 设置管道标志 */
    file_table[global_fd].fd_flag = PIPE_FLAG;  // 0xFFFF
    file_table[global_fd].fd_pos = 2;  // 两个描述符

    /* 安装文件描述符 */
    pipefd[0] = pcb_fd_install(global_fd);  // 读端
    pipefd[1] = pcb_fd_install(global_fd);  // 写端
    return 0;
}

/* 从管道读取数据 */
uint32_t pipe_read(int32_t fd, void* buf, uint32_t count) {
    char* buffer = buf;
    uint32_t bytes_read = 0;
    uint32_t global_fd = fd_local2global(fd);
    struct ioqueue* ioq = (struct ioqueue*)file_table[global_fd].fd_inode;

    /* 选择较小的数据量，避免阻塞 */
    uint32_t ioq_len = ioq_length(ioq);
    uint32_t size = ioq_len > count ? count : ioq_len;

    while (bytes_read < size) {
        *buffer = ioq_getchar(ioq);
        bytes_read++;
        buffer++;
    }
    return bytes_read;
}

/* 向管道写入数据 */
uint32_t pipe_write(int32_t fd, const void* buf, uint32_t count) {
    uint32_t bytes_write = 0;
    uint32_t global_fd = fd_local2global(fd);
    struct ioqueue* ioq = (struct ioqueue*)file_table[global_fd].fd_inode;

    /* 选择较小的数据量，避免阻塞 */
    uint32_t ioq_left = bufsize - ioq_length(ioq);
    uint32_t size = ioq_left > count ? count : ioq_left;

    const char* buffer = buf;
    while (bytes_write < size) {
        ioq_putchar(ioq, *buffer);
        bytes_write++;
        buffer++;
    }
    return bytes_write;
}

void my_shell(void) {
    cwd_cache[0] = '/';
    while (1) {
        print_prompt();
        memset(cmd_line, 0, MAX_PATH_LEN);
        readline(cmd_line, MAX_PATH_LEN);

        if (cmd_line[0] == 0) continue;

        /* 管道处理 */
        char* pipe_symbol = strchr(cmd_line, '|');
        if (pipe_symbol) {
            int32_t fd[2] = {-1};
            pipe(fd);  // 1. 创建管道

            /* 2. 处理第一个命令 */
            char* each_cmd = cmd_line;
            pipe_symbol = strchr(each_cmd, '|');
            *pipe_symbol = 0;  // 分割命令

            fd_redirect(1, fd[1]);  // 重定向标准输出到管道
            argc = cmd_parse(each_cmd, argv, ' ');
            cmd_execute(argc, argv);

            /* 3. 处理中间命令 */
            each_cmd = pipe_symbol + 1;
            fd_redirect(0, fd[0]);  // 重定向标准输入到管道

            while ((pipe_symbol = strchr(each_cmd, '|'))) {
                *pipe_symbol = 0;
                argc = cmd_parse(each_cmd, argv, ' ');
                cmd_execute(argc, argv);
                each_cmd = pipe_symbol + 1;
            }

            /* 4. 处理最后一个命令 */
            fd_redirect(1, 1);  // 恢复标准输出
            argc = cmd_parse(each_cmd, argv, ' ');
            cmd_execute(argc, argv);

            /* 5-6. 恢复标准输入并关闭管道 */
            fd_redirect(0, 0);
            close(fd[0]);
            close(fd[1]);
        } else {
            /* 普通命令处理 */
            argc = cmd_parse(cmd_line, argv, ' ');
            if (argc == -1) {
                printf("num of arguments exceed %d\n", MAX_ARG_NR);
                continue;
            }
            cmd_execute(argc, argv);
        }
    }
